[HttpPost("login")]
public async Task<IActionResult> Login([FromBody] LoginDto dto)
{
    // This check will fail if phone isn't confirmed
    var user = await _userManager.FindByEmailAsync(dto.Email);
    
    // When RequireConfirmedPhoneNumber = true, this returns false
    var canSignIn = await _signInManager.CanSignInAsync(user);
    
    if (!canSignIn)
        return Unauthorized("Phone number not confirmed");

    // If you check this way, it blocks JWT creation
    var result = await _signInManager.PasswordSignInAsync(dto.Email, dto.Password);
    if (!result.Succeeded) // Will fail if phone not confirmed
        return Unauthorized();
    
    // Token generation code won't execute
    var token = GenerateJwtToken(user);
}


[HttpPost("login")]
public async Task<IActionResult> Login([FromBody] LoginDto dto)
{
    // dto.PhoneNumber is the login identifier
    var user = await _userManager.FindByNameAsync(dto.PhoneNumber);
    
    if (user == null)
        return Unauthorized("Invalid phone number or password");

    // Check password
    if (!await _userManager.CheckPasswordAsync(user, dto.Password))
        return Unauthorized("Invalid phone number or password");

    // Manual phone confirmation check
    if (user.PhoneNumberConfirmed == false)
    {
        return Unauthorized(new 
        { 
            requiresPhoneConfirmation = true,
            message = "Phone number must be confirmed before login"
        });
    }

    // Generate JWT token
    var token = await GenerateJwtTokenAsync(user);
    return Ok(new { token, user = new { user.PhoneNumber, user.Id } });
}

[HttpPost("send-phone-verification")]
public async Task<IActionResult> SendPhoneVerificationCode()
{
    var user = await _userManager.GetUserAsync(User);
    if (user == null)
        return NotFound();

    // Generate phone number confirmation token
    var token = await _userManager.GenerateChangePhoneNumberTokenAsync(
        user, user.PhoneNumber);

    // Send via SMS (using Twilio, Azure Communication Services, etc.)
    await _smsService.SendSmsAsync(user.PhoneNumber, 
        $"Your verification code is: {token}");

    return Ok(new { message = "Verification code sent" });
}

[HttpPost("verify-phone")]
public async Task<IActionResult> VerifyPhoneNumber([FromBody] VerifyPhoneDto dto)
{
    var user = await _userManager.GetUserAsync(User);
    if (user == null)
        return NotFound();

    var isTokenValid = await _userManager.VerifyChangePhoneNumberTokenAsync(
        user, dto.Token, user.PhoneNumber);

    if (!isTokenValid)
        return BadRequest("Invalid verification code");

    user.PhoneNumberConfirmed = true;
    await _userManager.UpdateAsync(user);

    return Ok(new { message = "Phone number verified successfully" });
}
